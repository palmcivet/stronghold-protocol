```tsx [Code]
import { Select } from '@alliance/ui';

<Select
  label="语言"
  items={[{ id: 'zh', label: '中文' }]}
  selected="zh"
  onSelect={() => {}}
/>
```

:::

| 属性 | 类型 | 默认值 | 说明 |
| --- | --- | --- | --- |
| `label` | `string` | — | 无障碍名称 |
| `options` | `readonly { value: string; label: string }[]` | — | 可选项 |
| `value` | `string` | — | 当前值 |
| `onChange` | `(value: string) => void` | — | 选择变化时调用 |
| `isDisabled` | `boolean` | — | 禁用 |
